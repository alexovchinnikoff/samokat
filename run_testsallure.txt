@echo off
chcp 65001 >nul
title Запуск автотестов Яндекс Самокат

echo ==========================================
echo 🚀 Запуск автотестов проекта «Яндекс Самокат»
echo ==========================================

:: 1. Активируем виртуальное окружение
:: ВАЖНО: Мы ищем .venv в ТЕКУЩЕЙ папке (где лежит этот батник)
echo [1/4] Активация виртуального окружения...
call .\venv\Scripts\activate.bat

if %errorlevel% neq 0 (
    echo ❌ Ошибка: Не удалось активировать venv.
    echo Проверь, лежит ли папка .venv рядом с этим файлом.
    pause
    exit /b 1
)

:: 2. Запускаем pytest
echo [2/4] Запуск тестов и сбор данных...
:: --alluredir=allure-results: сохраняем данные для отчета
:: -v: подробный вывод имен тестов
python -m pytest --alluredir=allure-results -v

:: Сохраняем код возврата (прошли тесты или нет)
set TEST_RESULT=%errorlevel%

:: 3. Пытаемся открыть отчет Allure
echo [3/4] Попытка генерации отчета Allure...
:: Если у тебя установлен Allure CLI, он откроется в браузере.
:: Если нет - скрипт просто напишет ошибку, но тесты уже прошли!
call D:\Projects\samokat_root\samokat-third\tools\allure-2.45.0\bin\allure.bat serve allure-results



:: 4. Итог
echo [4/4] Готово!
if %TEST_RESULT% equ 0 (
    echo ==========================================
    echo 🎉 ВСЕ ТЕСТЫ ПРОЙДЕНЫ УСПЕШНО!
    echo ==========================================
) else (
    echo ==========================================
    echo ⚠️ ТЕСТЫ ЗАВЕРШЕНЫ С ОШИБКАМИ (см. консоль или отчет)
    echo ==========================================
)

pause
